Check Key

Gọi trực tiếp. Mọi lỗi trả cùng một message.

POST /functions/v1/check-key
Content-Type: application/json
apikey: 

{"key":"...","device_id":"...","app_version":"V1","ip":"optional"}

OK:  {"ok":true,"key_valid":true,"message":"Key hợp lệ."}
FAIL: {"ok":false,"key_valid":false,"message":"Key không hợp lệ."}

Java

OkHttpClient c = new OkHttpClient();
JSONObject b = new JSONObject();
b.put("key", key);
b.put("device_id", deviceId);
b.put("app_version", "V1");
Request req = new Request.Builder()
  .url(BASE + "/functions/v1/check-key")
  .addHeader("Content-Type", "application/json")
  .addHeader("apikey", API_KEY)
  .post(RequestBody.create(b.toString(), MediaType.parse("application/json")))
  .build();
try (Response res = c.newCall(req).execute()) {
  JSONObject j = new JSONObject(res.body().string());
  boolean ok = j.optBoolean("ok") && j.optBoolean("key_valid");
}

Python

import requests
r = requests.post(
  BASE + "/functions/v1/check-key",
  headers={"Content-Type":"application/json","apikey": API_KEY},
  json={"key": key, "device_id": device_id, "app_version": "V1"},
  timeout=10,
)
d = r.json()
ok = d.get("ok") and d.get("key_valid")

C++ / C (libcurl)

CURL *curl = curl_easy_init();
struct curl_slist *h = NULL;
h = curl_slist_append(h, "Content-Type: application/json");
h = curl_slist_append(h, "apikey: YOUR_KEY");
const char *body = "{\"key\":\"K\",\"device_id\":\"D\",\"app_version\":\"V1\"}";
curl_easy_setopt(curl, CURLOPT_URL, "https://PROJECT.supabase.co/functions/v1/check-key");
curl_easy_setopt(curl, CURLOPT_HTTPHEADER, h);
curl_easy_setopt(curl, CURLOPT_POSTFIELDS, body);
curl_easy_perform(curl);
curl_slist_free_all(h);
curl_easy_cleanup(curl);
