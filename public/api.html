Request req = new Request.Builder()
  .url("https://nklukqriopezsoalnghm.supabase.co/functions/v1/check-key")
  .addHeader("Content-Type", "application/json")
  .addHeader("apikey", "sb_publishable_RUE5iV8GqoVCFxjVt5ZBpg_FlTC1v-0")
  .post(RequestBody.create(body.toString(), MediaType.parse("application/json")))
  .build();

  JSONObject j = new JSONObject(res.body().string());
  boolean ok = j.optBoolean("ok", false) && j.optBoolean("key_valid", false);
}
